# T-ENCODE-03: behaviour of the bundled ffmpeg n8.1.2 (LGPL) measured while implementing design §4.
# Every command ran through a scratchpad wrapper (ff.mjs) that spawns the bundled ffmpeg/ffprobe and sets
# os.setPriority(PRIORITY_LOW) (idle). Inputs: lavfi, <= 2 s, <= 1280x720 (the RGB matrix check needs HD),
# made and deleted in the scratchpad. "avg" = mean signalstats YAVG/UAVG/VAVG of the first 5 frames.

## 1. RGB -> Y'CbCr: without an explicit matrix swscale converts with BT.601, whatever setparams writes
$ ffmpeg -f lavfi -i "color=red:s=1280x720:r=30:d=0.2,format=gbrp" -c:v ffv1 rgb.mkv      # probe: pix_fmt bgr0, color_space gbr, range pc
$ ffmpeg -i rgb.mkv -frames:v 1 -vf "format=yuv420p,setparams=color_primaries=bt709:color_trc=bt709:colorspace=bt709:range=tv" -c:v ffv1 rgb_auto.mkv
YAVG=81.00 UAVG=90.00 VAVG=240.00          (BT.601 red, tagged BT.709: a colour shift)
$ ffmpeg -i rgb.mkv -frames:v 1 -vf "scale=out_color_matrix=bt709:out_range=tv,format=yuv420p,setparams=...bt709..." -c:v ffv1 rgb_m.mkv
YAVG=63.00 UAVG=102.00 VAVG=240.00         (BT.709 red)
$ ffmpeg -i rgb.mkv -frames:v 1 -vf "scale=in_range=pc:out_range=tv:out_color_matrix=bt709,format=yuv420p" ...
YAVG=63.00 UAVG=102.00 VAVG=240.00         (in_range=pc is harmless on RGB input)
exit 0

## 2. Tone mapping followed by a scale: zscale must write Y'CbCr itself
TM = zscale=t=linear:npl=100,format=gbrpf32le,zscale=p=bt709,tonemap=tonemap=hable:desat=0,zscale=t=bt709:m=bt709:r=tv
$ ffmpeg -i hdr.mkv -frames:v 5 -vf "$TM,format=yuv420p" -c:v ffv1 t_noscale.mkv              -> YAVG=112.55 UAVG=138.47 VAVG=137.18
$ ffmpeg -i hdr.mkv -frames:v 5 -vf "$TM,scale=w=160:h=90:flags=...,format=yuv420p" ...        -> YAVG=107.97 UAVG=136.61 VAVG=133.10
  (debug log: Parsed_zscale_4 ... -> fmt:gbrpf32le; Parsed_scale_5 fmt:gbrpf32le -> yuv420p: swscale did the RGB->Y'CbCr)
$ ffmpeg -i hdr.mkv -frames:v 5 -vf "$TM,scale=w=320:h=180:flags=...,format=yuv420p" ...       -> YAVG=107.99 (same size: the conversion, not the resize)
$ ffmpeg -i hdr.mkv -frames:v 5 -vf "$TM,sidedata=...,sidedata=...,format=yuv420p,scale=w=160:h=90:flags=...,format=yuv420p,setparams=..." ...
  (debug log: Parsed_zscale_4 ... -> fmt:yuv420p csp:bt709 range:tv)                         -> YAVG=112.55 UAVG=138.44 VAVG=137.02
$ ffmpeg -i hdr.mkv -frames:v 5 -vf "$TM,sidedata=...,sidedata=...,format=yuv420p,format=nv12,setparams=..." ...   -> YAVG=112.55 UAVG=138.47 VAVG=137.18
exit 0

## 3. Tone mapping an untagged picture: zscale needs tin/pin/min AND p (output primaries default to the frame's)
$ ffmpeg -i untagged10.mkv -vf "$TM,format=yuv420p" -f null -
[Parsed_zscale_0] [error] code 3074: no path between colorspaces        exit 3752568763
$ ffmpeg -i untagged10.mkv -vf "zscale=t=linear:npl=100:tin=smpte2084:pin=bt2020:min=bt2020nc,format=gbrpf32le,..." -f null -
[Parsed_zscale_0] [error] code 3074: no path between colorspaces        exit 3752568763
$ ffmpeg -i untagged10.mkv -vf "zscale=t=linear:npl=100:tin=smpte2084:pin=bt2020:min=bt2020nc:p=bt2020,format=gbrpf32le,..." -frames:v 2 ...
YAVG=112.67 UAVG=138.35 VAVG=136.91        exit 0
$ (same picture tagged bt2020/smpte2084/bt2020nc with setparams, plain "$TM,format=yuv420p")
YAVG=112.67 UAVG=138.35 VAVG=136.91        exit 0   (identical to the explicit route)
